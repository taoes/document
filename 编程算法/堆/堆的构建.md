

## 新建堆（常规方式）

```java
package com.taoes.code.heap;

import java.util.*;

public class E堆的插入 {
    public static void main(String[] args) {
        int[] nums = new int[]{0, 100, 19, 36, 17, 3, 25, 1, 2};
        int[] heap = new int[nums.length];
        for (int i = 0; i < nums.length; i++) {
            heap[i] = nums[i];
            // 调整i 位的信息
            int t = i;
            while (t != 0) {
                int p = (t - 1) / 2;
                if (heap[p] < heap[i]) {
                    swap(heap, p, i);
                    t = p;
                    continue;
                }
                break;
            }
        }

        System.out.println(Arrays.toString(heap));
    }

    private static void swap(int[] heap, int p, int i) {
        int t = heap[p];
        heap[p] = heap[i];
        heap[i] = t;
    }
}
```

## 新建堆 (递归方式)

```java
package com.taoes.code.heap;  
  
import org.junit.Assert;  
  
import java.util.Arrays;  
  
public class E堆的构建 {  
    public static void main(String[] args) {  
        Solution solution = new Solution();  
        int[] nums = new int[]{0, 100, 19, 36, 17, 3, 25, 1, 2};  
        solution.buildMinHeap(nums);  
        System.out.println(Arrays.toString(nums));  
  
        solution.buildMaxHeap(nums);  
        System.out.println(Arrays.toString(nums));  
    }  
  
  
    static class Solution {  
  
        public void buildMinHeap(int[] nums) {  
            for (int i = nums.length / 2 - 1; i >= 0; i--) {  
                heapifyMin(nums, i, nums.length);  
            }  
        }  
  
        public void heapifyMin(int[] heap, int i, int len) {  
            int min = i;  
            int leftIndex = 2 * i + 1;  
            int rightIndex = 2 * i + 2;  
  
            if (leftIndex < len && heap[leftIndex] < heap[min]) {  
                min = leftIndex;  
            }  
  
            if (rightIndex < len && heap[rightIndex] < heap[min]) {  
                min = rightIndex;  
            }  
  
            if (min != i) {  
                swap(heap, i, min);  
                heapifyMin(heap, min, len);  
            }  
        }  
  
        public void buildMaxHeap(int[] nums) {  
            for (int i = nums.length / 2 - 1; i >= 0; i--) {  
                heapifyMax(nums, i, nums.length);  
            }  
  
        }  
  
        public void heapifyMax(int[] heap, int i, int len) {  
            int max = i;  
            int leftIndex = 2 * i + 1;  
            int rightIndex = 2 * i + 2;  
  
            if (leftIndex < len && heap[leftIndex] > heap[max]) {  
                max = leftIndex;  
            }  
  
            if (rightIndex < len && heap[rightIndex] > heap[max]) {  
                max = rightIndex;  
            }  
  
            if (max != i) {  
                swap(heap, i, max);  
                heapifyMax(heap, max, len);  
            }  
        }  
  
  
        public void swap(int[] heap, int p, int i) {  
            int t = heap[p];  
            heap[p] = heap[i];  
            heap[i] = t;  
        }  
    }  
}
```
