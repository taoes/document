## 贪心算法
  - 限制$0-3999$ 的情况下  空间复杂度:$O(1)$
  - 限制$0-3999$ 的情况下  时间复杂度:$O(1)$

```java
package com.taoes.code;

import org.junit.Assert;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.Map;

public class L03数字转罗马 {
    public static void main(String[] args) {
        Solution solution = new Solution();
        Assert.assertEquals("III", solution.romanToInt(3));
        Assert.assertEquals("VII", solution.romanToInt(7));
        Assert.assertEquals("XI", solution.romanToInt(11));

        Assert.assertEquals("IV", solution.romanToInt(4));
        Assert.assertEquals("IX", solution.romanToInt(9));
        Assert.assertEquals("MCMXCIV", solution.romanToInt(1994));
    }

    static class Solution {
        static Map<String, Integer> flagValue = new LinkedHashMap<>();


        static {
            flagValue.put("M", 1000);
            flagValue.put("CM", 900);
            flagValue.put("D", 500);
            flagValue.put("CD", 400);
            flagValue.put("C", 100);
            flagValue.put("XC", 90);
            flagValue.put("L", 50);
            flagValue.put("XL", 40);
            flagValue.put("X", 10);
            flagValue.put("IX", 9);
            flagValue.put("V", 5);
            flagValue.put("IV", 4);
            flagValue.put("I", 1);


            // 4,9,40,90,400,900


        }


        public String romanToInt(int num) {
            int tmp = num;
            ArrayList<Integer> integers = new ArrayList<>(flagValue.values());
            ArrayList<String> chares = new ArrayList<>(flagValue.keySet());
            StringBuilder build = new StringBuilder();
            while (tmp != 0) {
                for (int i = 0; i < integers.size(); i++) {
                    Integer value = integers.get(i);
                    if (tmp / value >= 1) {
                        build.append(chares.get(i));
                        tmp = tmp - value;
                    }
                }
            }
            return build.toString();
        }
    }
}
```
