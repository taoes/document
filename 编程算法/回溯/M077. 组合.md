
## 回溯

```java
package com.taoes.code.backtracking;

import lombok.var;

import java.util.*;

public class L77 {
    public static void main(String[] args) {
        Solution s = new Solution();
        var result = s.combine(4, 3);
        for (List<Integer> integers : result) {
            System.out.println(integers);
        }
    }


    static class Solution {
        final List<List<Integer>> combineResult = new ArrayList<>();
        final LinkedList<Integer> path = new LinkedList<>();

        public List<List<Integer>> combine(int n, int k) {
            this.combine(n, k, 1);
            return combineResult;
        }

        public void combine(int n, int k, int startIndex) {
            // 终止条件
            if (path.size() == k) {
                combineResult.add(new ArrayList<>(path));
                return;
            }
            // 按需遍历
            for (int i = startIndex; i <= n; i++) {
                path.push(i);
                this.combine(n, k, i + 1);
                path.pop();
            }
        }
    }
}
```
