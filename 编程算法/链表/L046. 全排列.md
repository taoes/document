给定一个不含重复数字的数组 `nums` ，返回其 _所有可能的全排列_ 。你可以 **按任意顺序** 返回答案。

## 回溯

```java
package com.taoes.code.backtracking;

import lombok.var;

import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;

public class L046 {
    public static void main(String[] args) {
        Solution s = new Solution();
        var result = s.permute(new int[]{1, 2, 3});
        for (List<Integer> integers : result) {
            System.out.println(integers);
        }
    }


    static class Solution {
        final List<List<Integer>> combineResult = new ArrayList<>();
        final LinkedList<Integer> path = new LinkedList<>();
        boolean[] used;

        public List<List<Integer>> permute(int[] nums) {
            used = new boolean[nums.length];
            this.combine(nums);
            return combineResult;
        }

        public void combine(int[] nums) {
            // 终止条件
            if (path.size() == nums.length) {
                combineResult.add(new ArrayList<>(path));
                return;
            }
            // 按需遍历
            for (int i = 0; i < nums.length; i++) {
                if (used[i]) {
                    continue;
                }
                used[i] = true;
                path.addLast(nums[i]);
                this.combine(nums);
                path.removeLast();
                used[i] = false;
            }
        }
    }
}
```
