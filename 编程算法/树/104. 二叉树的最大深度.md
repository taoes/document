


## 递归法

```java
package com.taoes.code.tree;  
  
import lombok.var;  
import org.junit.Assert;  
  
import java.util.Arrays;  
import java.util.LinkedList;  
import java.util.Queue;  
import java.util.Stack;  
  
/**  
 * 104. 二叉树的最大深度  
 */  
public class L104 {  
    public static void main(String[] args) {  
        var node = TreeNode.buildTree(new Integer[]{  
                3,  
                9, 20,  
                null, 15, 7  
        });  
  
  
        Assert.assertEquals(3, new Solution().maxDepth(node));  
    }  
  
    static class Solution {  
        public int maxDepth(TreeNode root) {  
            if (root == null) {  
                return 0;  
            }  
            int maxDepthLeft = this.maxDepth(root.left);  
            int maxDepthRight = this.maxDepth(root.right);  
            return Math.max(maxDepthLeft, maxDepthRight) + 1;  
        }  

    }  
  
  
}
```


## 迭代法

```java
package com.taoes.code.tree;  
  
import lombok.var;  
import org.junit.Assert;  
  
import java.util.Arrays;  
import java.util.LinkedList;  
import java.util.Queue;  
import java.util.Stack;  
  
/**  
 * 104. 二叉树的最大深度  
 */  
public class L104 {  
    public static void main(String[] args) {  
        var node = TreeNode.buildTree(new Integer[]{  
                3,  
                9, 20,  
                null, 15, 7  
        });  
  
  
        Assert.assertEquals(3, new Solution().maxDepth2(node));  
    }  
  
    static class Solution {  
      
        public int maxDepth2(TreeNode root) {  
            if (root == null) {  
                return 0;  
            }  
            int depth = 0;  
  
            Queue<TreeNode> queue = new LinkedList<>();  
            queue.add(root);  
  
  
            while (!queue.isEmpty()) {  
                depth++;  
  
                Queue<TreeNode> next_queue = new LinkedList<>();  
                for (TreeNode treeNode : queue) {  
                    if (treeNode.left != null) {  
                        next_queue.add(treeNode.left);  
                    }  
  
                    if (treeNode.right != null) {  
                        next_queue.add(treeNode.right);  
                    }  
                }  
  
                queue = next_queue;  
            }  
            return depth;  
        }  
    }  
  
  
}
```
